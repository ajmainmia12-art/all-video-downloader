// UNIVERSAL DOWNLOADER PRO - SINGLE FILE - 70 COUNTRY SUPPORT // File: lib/main.dart - এই একটা ফাইলেই সব! import 'package:flutter/material.dart'; import 'package:flutter_inappwebview/flutter_inappwebview.dart'; import 'package:dio/dio.dart'; import 'package:path_provider/path_provider.dart'; import 'package:permission_handler/permission_handler.dart'; void main() async { WidgetsFlutterBinding.ensureInitialized(); await Permission.storage.request(); await Permission.manageExternalStorage.request(); runApp(const UniversalDownloaderApp()); } class UniversalDownloaderApp extends StatelessWidget { const UniversalDownloaderApp({super.key}); @override Widget build(BuildContext context) { return MaterialApp( title: 'Downloader Pro', debugShowCheckedModeBanner: false, theme: ThemeData(useMaterial3: true, colorSchemeSeed: Colors.blue), home: const SingleFileDownloader(), ); } } class SingleFileDownloader extends StatefulWidget { const SingleFileDownloader({super.key}); @override State createState() => _SingleFileDownloaderState(); } class _SingleFileDownloaderState extends State { final TextEditingController urlCtrl = TextEditingController(); InAppWebViewController?
